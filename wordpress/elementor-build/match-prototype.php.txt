/* ---------------- a reusable "shop these" rail, styled like the home rails ---------------- */
$railProducts = $find($home, '7b0eb87');
$makeRail = function ($eyebrow, $title, $ids, $anchor) use ($find, $reid, $newid, $railProducts, $home) {
  $head = $find($home, '515977e');           // the Best sellers section, for its heading block
  $section = $reid($head);
  $section['settings']['_element_id'] = $anchor;
  $products = $reid($railProducts);
  $products['settings']['query_post_type'] = 'by_id';
  $products['settings']['query_posts_ids'] = $ids;
  unset($products['settings']['orderby'], $products['settings']['order']);
  $walk = function (&$els) use (&$walk, $eyebrow, $title, $products, &$done) {
    foreach ($els as $i => &$e) {
      $t = $e['widgetType'] ?? '';
      if ($t === 'heading') {
        $current = $e['settings']['title'] ?? '';
        if (strpos($current, 'Moving fast') !== false) { $e['settings']['title'] = $eyebrow; }
        elseif ($current === 'Best sellers') { $e['settings']['title'] = $title; }
      }
      if ($t === 'woocommerce-products') { $e = $products; }
      if (!empty($e['elements'])) { $walk($e['elements']); }
    }
  };
  $walk($section['elements']);
  return $section;
};
